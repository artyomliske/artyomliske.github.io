def occupied_interval(
    starts_at: datetime,
    ends_at: datetime,
    buffer_before_min: int,
    buffer_after_min: int,
) -> Interval:
    # Интервал, который бронь реально занимает с учётом буферов (правило 4.1).
    # Именно по нему БД проверяет пересечения, а не по времени съёмки.
    ensure_aware(starts_at, "starts_at")
    ensure_aware(ends_at, "ends_at")
    if ends_at <= starts_at:
        raise InvalidInterval("ends_at должен быть строго позже starts_at")
    if buffer_before_min < 0 or buffer_after_min < 0:
        raise InvalidInterval("Буферы не могут быть отрицательными")

    return Interval(
        start=starts_at - timedelta(minutes=buffer_before_min),
        end=ends_at + timedelta(minutes=buffer_after_min),
    )

export function autoSeat(
  tables: string[],
  entries: AutoSeatEntry[],
  seatsPerTable = SEATS_PER_TABLE,
): SeatAssignment[] {
  const occupancy = new Map<string, Set<number>>();
  for (const t of tables) occupancy.set(t, new Set<number>());

  for (const e of entries) {
    if (e.status === "in" && e.table && e.seat != null) {
      occupancy.get(e.table)?.add(e.seat);
    }
  }

  const assignments: SeatAssignment[] = [];
  const unseated = entries.filter((e) => e.status === "in" && !e.table);

  for (const e of unseated) {
    // каждый следующий садится за наименее занятый стол
    let bestTable: string | null = null;
    let bestCount = Infinity;
    for (const t of tables) {
      const occ = occupancy.get(t);
      if (occ && occ.size < seatsPerTable && occ.size < bestCount) {
        bestTable = t;
        bestCount = occ.size;
      }
    }
    if (!bestTable) break; // свободных мест не осталось

    const occ = occupancy.get(bestTable)!;
    let seat = 1;
    while (occ.has(seat)) seat++;
    occ.add(seat);
    assignments.push({ playerId: e.playerId, table: bestTable, seat });
  }
  return assignments;
}

Это та самая функция из проекта — в браузере выполняется она сама, без упрощений: столы заполняются по наименее занятому. This is the project’s own function, running here unmodified: every next player takes a seat at the least-occupied table.

export const DEFAULT_BASE_POINTS = 10;

/**
 * Очки за место по авто-формуле: base · √N / √place.
 * Награждает за крупные турниры и высокие места.
 */
export function autoPoints(
  place: number,
  fieldSize: number,
  base = DEFAULT_BASE_POINTS,
): number {
  const n = Math.max(1, Math.floor(Number(fieldSize) || 1));
  const p = Math.max(1, Math.floor(Number(place) || 1));
  const b = Number(base) || DEFAULT_BASE_POINTS;
  return Math.round((b * Math.sqrt(n)) / Math.sqrt(p));
}

Формула клуба целиком: крупный турнир весит больше, но второе место никогда не догонит первое. Столбики — очки за места с 1-го по 12-е при выбранном поле. The club’s whole formula: a bigger field weighs more, yet second place never catches first. The bars are points for places 1–12 at the chosen field size.

/** Площадь проёмов (окна + двери), м² */
function openingsArea(a: WizardAnswers): number {
  return (a.windows || 0) * 1.8 + (a.doorways || 0) * 2.0;
}

/** Чистая площадь стен за вычетом проёмов
 *  (не опускается ниже 50% от валовой) */
export function wallArea(a: WizardAnswers): number {
  const gross = a.perimeter * a.ceilingHeight;
  const net = gross - openingsArea(a);
  return Math.max(net, gross * 0.5);
}

Из этих метров считается вся малярка и штукатурка сметы. Нижняя граница в 50% — защита от «периметра на глаз»: смета не обнулится, если менеджер напутал с проёмами. All plastering and painting in the estimate is derived from these square metres. The 50% floor guards against a guessed perimeter: the estimate cannot collapse to nothing if the manager miscounts openings.

    Как я работаюHow I work

    1. Разбираем процесс и пишем ТЗWe map the process and write the spec

      До первой строчки кода на бумаге появляется, что система делает, чего она делать не будет и по каким правилам считает деньги. Это дешевле любой переделки.

      Before the first line of code, it is written down what the system does, what it deliberately will not do, and by which rules it counts money. That is cheaper than any rework.

    2. Собираем этапамиWe build in stages

      Каждый этап — работающая часть, которую можно открыть и потрогать, а не обещание к концу проекта. Приоритет у того, что снимает боль сегодня.

      Each stage is a working piece you can open and use, not a promise for the end of the project. Whatever removes today’s pain goes first.

    3. Правила защищаем проверкамиRules are protected by checks

      Ключевые правила — занятость слота, расчёт цены, судьба предоплаты — закрыты автоматическими тестами, которые прогоняются на каждое изменение. «Вроде работает» заменяется на «проверено».

      The critical rules — slot availability, price calculation, what happens to a deposit — are covered by automated tests that run on every change. “Seems to work” becomes “verified”.

    4. Ставим в работу и остаёмся на связиWe go live and I stay reachable

      Сервер, домен, перенос старых данных, обучение сотрудников. Дальше — доработки по мере того, как процесс меняется, потому что он меняется.

      Server, domain, migration of old data, training the team. Then changes as the process evolves — because it will.

    Чего я не делаюWhat I don’t take on

    • Не берусь, если задачу закрывает готовая коробкаI pass when an off-the-shelf tool already solves it

      Если процесс укладывается в готовую CRM или в таблицу — скажу это на первом разговоре. Своя система оправдана тогда, когда процесс не влезает в чужую логику, а не тогда, когда хочется своё.

      If your process fits an existing CRM or even a spreadsheet, I will say so in the first conversation. A custom system earns its cost when the process does not fit someone else’s logic — not when custom simply sounds better.

    • Не переписываю большой чужой код с нуляI don’t rewrite a large existing codebase from scratch

      Могу встроиться рядом и забрать на себя часть процесса. А «перепишем всё за месяц» — обещание, которое почти никогда не сбывается, и я его не даю.

      I can build alongside it and take over part of the process. “We’ll rewrite everything in a month” is a promise that almost never holds, so I don’t make it.

    • Не рисую макеты отдельно от разработкиI don’t deliver mock-ups detached from the build

      Интерфейс собираю сразу в коде — его можно открыть и потрогать на своём телефоне, а не смотреть на картинку и додумывать, как это себя поведёт.

      I build the interface directly in code — you can open it on your own phone instead of studying a picture and guessing how it would behave.

    • Не начинаю без письменного ТЗI don’t start without a written spec

      Пока не записано, что система делает и чего она делать не будет, любая оценка сроков — выдумка. Это не бюрократия, а способ не платить дважды.

      Until it is written down what the system does and what it deliberately will not do, any estimate is fiction. That is not bureaucracy — it is how you avoid paying twice.

    • Не занимаюсь трафиком и маркетингомI don’t do marketing or lead generation

      Собираю систему, которая держит процесс и деньги. Приводить клиентов и вести рекламу — не ко мне, и лучше вы услышите это сейчас, чем через месяц.
